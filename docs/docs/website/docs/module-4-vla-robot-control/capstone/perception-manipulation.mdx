# Perception and Manipulation

This section covers the perception and manipulation flow in Vision-Language-Action (VLA) systems, focusing on how visual information is processed and used for robotic manipulation tasks.

## Overview

Perception and manipulation are critical components of the VLA pipeline that enable robots to understand their environment and interact with objects. Perception processes visual information to identify objects and their locations, while manipulation executes physical actions to interact with these objects.

## Technical Architecture

The perception and manipulation flow follows this architecture:

1. Visual input from cameras or sensors
2. Object detection and recognition
3. Spatial reasoning and localization
4. Manipulation planning
5. Robot arm control
6. Grasp verification
7. Feedback and confirmation

## Implementation Steps

### 1. Perception Pipeline

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, CameraInfo
from std_msgs.msg import String
from geometry_msgs.msg import Point
from cv_bridge import CvBridge
import cv2
import numpy as np
import json

class PerceptionPipeline(Node):
    def __init__(self):
        super().__init__('perception_pipeline')

        # Publishers
        self.object_detection_publisher = self.create_publisher(String, 'detected_objects', 10)
        self.perception_result_publisher = self.create_publisher(String, 'perception_results', 10)

        # Subscribers
        self.image_subscriber = self.create_subscription(
            Image,
            'camera/image_raw',
            self.image_callback,
            10
        )

        self.intent_subscriber = self.create_subscription(
            String,
            'robot_intents',
            self.intent_callback,
            10
        )

        # Initialize CV bridge
        self.bridge = CvBridge()

        # Object detection model parameters
        self.confidence_threshold = 0.5
        self.target_objects = ['cup', 'bottle', 'box', 'chair', 'table']

        # Camera parameters (would be loaded from camera info in real implementation)
        self.camera_matrix = np.array([[554.25, 0.0, 320.5], [0.0, 554.25, 240.5], [0.0, 0.0, 1.0]])

    def intent_callback(self, msg):
        """Handle perception intents"""
        try:
            intent_data = json.loads(msg.data)
            intent_type = intent_data.get('type', '')

            if intent_type == 'sensing':
                target = intent_data.get('parameters', {}).get('target', '')
                self.get_logger().info(f'Received sensing intent for: {target}')

                # Set target for object detection
                self.target_objects = [target] if target else self.target_objects

        except json.JSONDecodeError as e:
            self.get_logger().error(f'Invalid JSON in intent: {e}')

    def image_callback(self, msg: Image):
        """Process incoming images for object detection"""
        try:
            # Convert ROS image to OpenCV image
            cv_image = self.bridge.imgmsg_to_cv2(msg, desired_encoding='bgr8')

            # Perform object detection
            detections = self.detect_objects(cv_image)

            # Publish detected objects
            if detections:
                detection_msg = String()
                detection_msg.data = json.dumps({
                    'objects': detections,
                    'timestamp': msg.header.stamp.sec,
                    'frame_id': msg.header.frame_id
                })
                self.object_detection_publisher.publish(detection_msg)

                self.get_logger().info(f'Detected {len(detections)} objects')

        except Exception as e:
            self.get_logger().error(f'Error processing image: {e}')

    def detect_objects(self, image):
        """Detect objects in the image using color and shape analysis"""
        detections = []

        # Convert BGR to HSV for color detection
        hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)

        # Define color ranges for common objects
        color_ranges = {
            'red_cup': ([0, 50, 50], [10, 255, 255]),
            'blue_bottle': ([100, 50, 50], [130, 255, 255]),
            'green_bottle': ([40, 50, 50], [80, 255, 255]),
            'yellow_box': ([20, 50, 50], [30, 255, 255])
        }

        for obj_name, (lower, upper) in color_ranges.items():
            lower = np.array(lower, dtype="uint8")
            upper = np.array(upper, dtype="uint8")

            # Create mask
            mask = cv2.inRange(hsv, lower, upper)

            # Apply morphological operations to reduce noise
            kernel = np.ones((5,5), np.uint8)
            mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
            mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)

            # Find contours
            contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

            for contour in contours:
                area = cv2.contourArea(contour)
                if area > 1000:  # Filter out small areas
                    # Calculate center of the object
                    M = cv2.moments(contour)
                    if M["m00"] != 0:
                        cX = int(M["m10"] / M["m00"])
                        cY = int(M["m01"] / M["m00"])

                        # Estimate depth (simplified - in real implementation use depth camera)
                        # For now, just use inverse of area as a proxy for distance
                        distance = 1000.0 / np.sqrt(area) if area > 0 else float('inf')

                        detections.append({
                            'name': obj_name,
                            'center': (cX, cY),
                            'area': area,
                            'distance': distance,
                            'confidence': min(1.0, area / 5000.0)  # Normalize confidence
                        })

        # Filter detections based on target objects if specified
        if hasattr(self, 'target_objects') and self.target_objects:
            filtered_detections = [d for d in detections if any(target in d['name'] for target in self.target_objects)]
            return filtered_detections

        return detections

    def estimate_3d_position(self, pixel_x, pixel_y, distance):
        """Estimate 3D position from 2D pixel coordinates and distance"""
        # Convert 2D pixel coordinates to 3D world coordinates
        # This is a simplified version - in real implementation use camera calibration
        fx = self.camera_matrix[0, 0]
        fy = self.camera_matrix[1, 1]
        cx = self.camera_matrix[0, 2]
        cy = self.camera_matrix[1, 2]

        # Calculate 3D position relative to camera
        x = (pixel_x - cx) * distance / fx
        y = (pixel_y - cy) * distance / fy
        z = distance

        return (x, y, z)

def main(args=None):
    rclpy.init(args=args)
    perception_pipeline = PerceptionPipeline()
    rclpy.spin(perception_pipeline)
    perception_pipeline.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```

### 2. Manipulation Manager

```python
import rclpy
from rclpy.node import Node
from std_msgs.msg import String
from geometry_msgs.msg import Pose, Point
from sensor_msgs.msg import JointState
import json

class ManipulationManager(Node):
    def __init__(self):
        super().__init__('manipulation_manager')

        # Publishers
        self.manipulation_result_publisher = self.create_publisher(String, 'manipulation_results', 10)
        self.arm_command_publisher = self.create_publisher(String, 'arm_commands', 10)

        # Subscribers
        self.intent_subscriber = self.create_subscription(
            String,
            'robot_intents',
            self.intent_callback,
            10
        )

        self.object_subscriber = self.create_subscription(
            String,
            'detected_objects',
            self.object_callback,
            10
        )

        self.joint_state_subscriber = self.create_subscription(
            JointState,
            'joint_states',
            self.joint_state_callback,
            10
        )

        # Manipulation state
        self.joint_positions = {}
        self.detected_objects = []
        self.target_object = None

    def intent_callback(self, msg):
        """Handle manipulation intents"""
        try:
            intent_data = json.loads(msg.data)
            intent_type = intent_data.get('type', '')

            if intent_type == 'manipulation':
                target = intent_data.get('parameters', {}).get('target', '')
                self.get_logger().info(f'Received manipulation intent for: {target}')

                # Store target object
                self.target_object = target

        except json.JSONDecodeError as e:
            self.get_logger().error(f'Invalid JSON in intent: {e}')

    def object_callback(self, msg):
        """Handle detected objects"""
        try:
            objects_data = json.loads(msg.data)
            self.detected_objects = objects_data.get('objects', [])

            # If we have a target object and detected objects, try to manipulate
            if self.target_object and self.detected_objects:
                self.attempt_manipulation()

        except json.JSONDecodeError as e:
            self.get_logger().error(f'Invalid JSON in object detection: {e}')

    def attempt_manipulation(self):
        """Attempt to manipulate the target object"""
        # Find the target object in detected objects
        target_obj = None
        for obj in self.detected_objects:
            if self.target_object.lower() in obj['name'].lower():
                target_obj = obj
                break

        if target_obj:
            self.get_logger().info(f'Found target object: {target_obj["name"]}')

            # Plan manipulation trajectory
            success = self.plan_and_execute_manipulation(target_obj)

            # Publish result
            result_msg = String()
            result_msg.data = json.dumps({
                'target': self.target_object,
                'success': success,
                'action': 'manipulation',
                'object_info': target_obj
            })
            self.manipulation_result_publisher.publish(result_msg)
        else:
            self.get_logger().warn(f'Target object "{self.target_object}" not found in detected objects')

    def plan_and_execute_manipulation(self, obj_info):
        """Plan and execute manipulation for the given object"""
        self.get_logger().info(f'Planning manipulation for {obj_info["name"]}')

        try:
            # Calculate target pose for manipulation
            # In a real implementation, this would involve:
            # - Inverse kinematics
            # - Collision checking
            # - Trajectory planning

            # For this example, we'll use a simplified approach
            target_pose = self.calculate_manipulation_pose(obj_info)

            if target_pose:
                # Execute manipulation
                success = self.execute_manipulation(target_pose)
                return success
            else:
                self.get_logger().error('Could not calculate manipulation pose')
                return False

        except Exception as e:
            self.get_logger().error(f'Error in manipulation planning: {e}')
            return False

    def calculate_manipulation_pose(self, obj_info):
        """Calculate the manipulation pose for the object"""
        # This is a simplified calculation
        # In a real implementation, this would use inverse kinematics
        try:
            # Get object position in 3D space
            # For now, we'll use a simplified conversion
            x = obj_info['center'][0] * 0.01  # Convert pixel to meters (simplified)
            y = obj_info['center'][1] * 0.01  # Convert pixel to meters (simplified)
            z = obj_info['distance'] * 0.01   # Convert to meters (simplified)

            # Create target pose
            pose = {
                'position': {'x': x, 'y': y, 'z': z + 0.1},  # Above the object
                'orientation': {'x': 0.0, 'y': 0.0, 'z': 0.0, 'w': 1.0}
            }

            return pose
        except Exception as e:
            self.get_logger().error(f'Error calculating manipulation pose: {e}')
            return None

    def execute_manipulation(self, target_pose):
        """Execute the manipulation action"""
        self.get_logger().info(f'Executing manipulation to pose: {target_pose}')

        try:
            # In a real implementation, this would:
            # - Send commands to robot arm
            # - Monitor execution
            # - Verify grasp success

            # For this example, we'll simulate execution
            command = {
                'action': 'move_to_pose',
                'pose': target_pose,
                'gripper': 'open'  # Open gripper initially
            }

            cmd_msg = String()
            cmd_msg.data = json.dumps(command)
            self.arm_command_publisher.publish(cmd_msg)

            # Simulate grasp
            grasp_command = {
                'action': 'grasp',
                'gripper': 'close'
            }

            cmd_msg.data = json.dumps(grasp_command)
            self.arm_command_publisher.publish(cmd_msg)

            # For this example, assume success
            return True

        except Exception as e:
            self.get_logger().error(f'Error executing manipulation: {e}')
            return False

    def joint_state_callback(self, msg: JointState):
        """Update joint positions"""
        for i, name in enumerate(msg.name):
            self.joint_positions[name] = msg.position[i]

def main(args=None):
```

### 3. Perception-Manipulation Integration

```python
import rclpy
from rclpy.node import Node
from std_msgs.msg import String
import json

class PerceptionManipulationIntegration(Node):
    def __init__(self):
        super().__init__('perception_manipulation_integration')

        # Publishers
        self.integration_result_publisher = self.create_publisher(String, 'integration_results', 10)

        # Subscribers
        self.perception_result_subscriber = self.create_subscription(
            String,
            'perception_results',
            self.perception_result_callback,
            10
        )

        self.manipulation_result_subscriber = self.create_subscription(
            String,
            'manipulation_results',
            self.manipulation_result_callback,
            10
        )

        # State tracking
        self.perception_results = {}
        self.manipulation_results = {}

    def perception_result_callback(self, msg):
        """Handle perception results"""
        try:
            result = json.loads(msg.data)
            self.perception_results = result

            self.get_logger().info(f'Received perception result: {result}')

            # Check if we have both perception and manipulation results
            self.check_integration_completion()

        except json.JSONDecodeError as e:
            self.get_logger().error(f'Invalid JSON in perception result: {e}')

    def manipulation_result_callback(self, msg):
        """Handle manipulation results"""
        try:
            result = json.loads(msg.data)
            self.manipulation_results = result

            self.get_logger().info(f'Received manipulation result: {result}')

            # Check if we have both perception and manipulation results
            self.check_integration_completion()

        except json.JSONDecodeError as e:
            self.get_logger().error(f'Invalid JSON in manipulation result: {e}')

    def check_integration_completion(self):
        """Check if both perception and manipulation are complete"""
        if self.perception_results and self.manipulation_results:
            # Combine results
            integration_result = {
                'perception': self.perception_results,
                'manipulation': self.manipulation_results,
                'integration_status': 'completed',
                'timestamp': self.get_clock().now().to_msg().sec
            }

            # Publish integration result
            result_msg = String()
            result_msg.data = json.dumps(integration_result)
            self.integration_result_publisher.publish(result_msg)

            self.get_logger().info('Perception-manipulation integration completed')

            # Clear results for next cycle
            self.perception_results = {}
            self.manipulation_results = {}

def main(args=None):
    rclpy.init(args=args)
    integration_node = PerceptionManipulationIntegration()
    rclpy.spin(integration_node)
    integration_node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
```

## Best Practices

- Use multiple sensors for robust perception (RGB, depth, etc.)
- Implement proper calibration for accurate 3D position estimation
- Plan manipulation trajectories carefully to avoid collisions
- Verify grasp success with force/torque sensors
- Handle occlusions and dynamic environments gracefully

## Troubleshooting

Common issues and solutions:

- **Object detection failures**: Improve lighting conditions and use multiple detection methods
- **Grasp failures**: Implement grasp planning with multiple grasp points
- **Coordinate frame issues**: Ensure consistent coordinate system usage
- **Real-time performance**: Optimize algorithms for real-time execution
